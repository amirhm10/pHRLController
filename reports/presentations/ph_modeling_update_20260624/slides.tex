\frametitle{\large \textbf{The Acetic-Acid/Acetate Pair Is the Buffer}}
\small

\vspace{-0.10cm}
\begin{columns}[T,onlytextwidth]

\begin{column}{0.45\textwidth}
\begin{block}{Source-document terminology}
\begin{itemize}\raggedright
  \setlength\itemsep{0.55ex}
  \item Flow 0: acetic acid, $\FH$
  \item Flow 1: sodium acetate, $\FA$
  \item Flow 2: water, $\FW$
  \item Acid and acetate stocks: about 100 mM
\end{itemize}
\end{block}

\vspace{0.05cm}
\begin{block}{Main modeling split}
\justifying
Separate the \textbf{chemical pH effect} from the \textbf{hydraulic/dynamic effect}.
\end{block}
\end{column}

\begin{column}{0.52\textwidth}
\begin{block}{Henderson-Hasselbalch baseline}
\vspace{-0.15cm}
\[
  \pH \approx pK_a+\log_{10}\frac{C_{A^-}}{C_{HA}}
\]
\vspace{-0.18cm}
\[
  C_{HA}=\frac{C_0\FH}{\FH+\FA+\FW},
  \qquad
  C_{A^-}=\frac{C_0\FA}{\FH+\FA+\FW}
\]
\vspace{-0.18cm}
\[
  \pH \approx pK_a+\log_{10}\frac{\FA}{\FH}
\]
\vspace{-0.18cm}
\end{block}

\vspace{0.05cm}
\begin{beamercolorbox}[rounded=true,center,sep=4pt,wd=0.96\linewidth]{block body}
{\bfseries The actual pH actuator is mainly the sodium-acetate/acetic-acid ratio.}
\end{beamercolorbox}
\end{column}
\end{columns}
\end{frame}

%============================= Slide 4 ============================
\begin{frame}
\setbeamertemplate{blocks}[rounded][shadow=false]
\setbeamercolor{block body}{fg=black,bg=blockTint}
\frametitle{\large \textbf{Water Cancels From Ideal pH, But Not From Dynamics}}
\small

\vspace{-0.12cm}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.47\textwidth}
\begin{block}{Why water is weak for steady pH}
\justifying
In the acetate-buffer region, water is a weak steady-state pH input. The model should not force water to explain pH changes.
\end{block}

\vspace{0.05cm}
\begin{block}{What the data check found}
\begin{itemize}\raggedright
  \setlength\itemsep{0.45ex}
  \item Charge-balance model includes dilution by water.
  \item Difference from HH: about $0.001$ pH on average.
  \item Maximum checked difference: less than $0.009$ pH.
\end{itemize}
\end{block}
\end{column}

\begin{column}{0.50\textwidth}
\begin{block}{How water still matters}
\vspace{-0.12cm}
\[
  \Cbuf=\frac{C_0(\FH+\FA)}{\FT},
  \qquad
  \FT=\FH+\FA+\FW
\]
\vspace{-0.18cm}
\[
  \tau \approx \frac{V_{\mathrm{eff}}}{\FT}
\]
\vspace{-0.15cm}
\begin{itemize}\raggedright
  \setlength\itemsep{0.38ex}
  \item dilution and buffer strength
  \item total flow and residence time
  \item mixing delay and PH\_2 sensor response
  \item activity coefficients and apparent pKa
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}

\begin{frame}
\setbeamertemplate{blocks}[rounded][shadow=false]

\frametitle{\large \textbf{Apparent pKa Is a Lumped Intercept Diagnostic}}
\small

\vspace{-0.12cm}
\begin{columns}[T,onlytextwidth]

\begin{column}{0.48\textwidth}
\begin{block}{Calculation}
\vspace{-0.10cm}
\[
  pK_{a,\mathrm{app},k}
  =
  \pH_k-\log_{10}
  \left(\frac{C_{acetate,0}F_{acetate,k}}
  {C_{acid,0}F_{acid,k}}\right)
\]
\vspace{-0.20cm}
\[
  C_{acid,0}=C_{acetate,0}
  \quad\Rightarrow\quad
  pK_{a,\mathrm{app},k}
  =
  \pH_k-\log_{10}\frac{F_{acetate,k}}{F_{acid,k}}
\]
\vspace{-0.20cm}
\end{block}

\vspace{0.05cm}
\begin{block}{Not the pure textbook pKa}
\justifying
It includes sensor bias, activity effect, temperature effect, stock concentration error, and mixing/delay error.
\end{block}
\end{column}

\begin{column}{0.49\textwidth}
\begin{block}{What changed in the data}
\begin{tabular}{@{}lcc@{}}
\toprule
segment & residual & $pK_{a,\mathrm{app}}$\\
\midrule
0--182 & -0.037 & 4.723\\
183--308 & -0.336 & 4.424\\
309--961 & -0.347 & 4.413\\
\bottomrule
\end{tabular}
\end{block}

\vspace{0.05cm}
\begin{beamercolorbox}[rounded=true,center,sep=4pt,wd=0.96\linewidth]{block body}
{\bfseries Before sample 183 looks reasonable; after sample 183 is a different intercept regime.}
\end{beamercolorbox}
\end{column}
\end{columns}
\end{frame}

\frametitle{\large \textbf{The Post-183 Apparent pKa Is Not a Temperature pKa}}
\small

\vspace{-0.12cm}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\begin{block}{Temperature explanation check}
\begin{itemize}\raggedright
  \setlength\itemsep{0.55ex}
  \item Literature pKa near room temperature is about 4.76.
  \item The source document notes normal aqueous temperature dependence stays near this range.
  \item The post-183 value is about 4.41.
  \item This is too large to be normal temperature variation.
\end{itemize}
\end{block}
\end{column}

\begin{column}{0.49\textwidth}
\begin{block}{Equivalent ratio interpretation}
\vspace{-0.10cm}
\[
  10^{-0.336}\approx 0.46
\]
\vspace{-0.18cm}
\justifying
After sample 183, the data behave as if the effective acetate/acid strength ratio were about half of the assumed value, or as if the reliable pH sensor had an offset of about $-0.3$ pH.
\end{block}

\vspace{0.05cm}
\begin{beamercolorbox}[rounded=true,center,sep=4pt,wd=0.96\linewidth]{block body}
{\bfseries Interpret this as a session-dependent calibration/intercept shift.}
\end{beamercolorbox}
\end{column}
\end{columns}
\end{frame}

%============================= Slide 11 ============================
\begin{frame}
\setbeamertemplate{blocks}[rounded][shadow=false]
\setbeamercolor{block body}{fg=black,bg=blockTint}
\frametitle{\large \textbf{Raw Three-Flow Control Is Not a Unique pH Problem}}
\small

\vspace{-0.10cm}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.45\textwidth}
\begin{block}{Direct raw action}
\vspace{-0.12cm}
\[
  u=\begin{bmatrix}F_{acid}\\F_{acetate}\end{bmatrix}
\]
\vspace{-0.20cm}
\[
  \pH \approx pK_a+\log_{10}\frac{F_{acetate}}{F_{acid}}
\]
\vspace{-0.16cm}
\end{block}

\vspace{0.05cm}
\begin{block}{Same pH, different usage}
\[
  (1,1),\quad (3,3),\quad (8,8)
  \quad\Rightarrow\quad
  \frac{F_{acetate}}{F_{acid}}=1
\]
\end{block}
\end{column}

\begin{column}{0.52\textwidth}
\begin{block}{Document warning}
\begin{itemize}\raggedright
  \setlength\itemsep{0.45ex}
  \item The pH-tracking reward is not uniquely defined.
  \item RL can drift to unnecessarily high flows.
  \item RL can drift to very low flows.
  \item Pump actions can chatter.
  \item Water or total flow can look useful for a pH problem.
\end{itemize}
\end{block}

\vspace{0.05cm}
\begin{beamercolorbox}[rounded=true,center,sep=4pt,wd=0.96\linewidth]{block body}
{\bfseries Do not let RL freely choose acid and acetate independently for the first controller.}
\end{beamercolorbox}
\end{column}
\end{columns}
\end{frame}

\begin{frame}

\frametitle{\large \textbf{Controller Structure: Ratio-Based Controller}}
\scriptsize

\vspace{-0.12cm}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.38\textwidth}
\begin{block}{Reparameterize the action}
\vspace{-0.12cm}
\[
  r=\frac{F_{acetate}}{F_{acid}},
  \qquad
  z=\log_{10}\frac{F_{acetate}}{F_{acid}}
\]
\vspace{-0.20cm}
\[
  \pH \approx pK_a+z
\]
\vspace{-0.16cm}
\[
  F_B=F_{acid}+F_{acetate}
\]
\vspace{-0.16cm}
\end{block}

\vspace{0.05cm}
\begin{block}{Deterministic flow allocator}
\vspace{-0.12cm}
\[
  r=10^z,\quad
  F_{acid}=\frac{F_B}{1+r},\quad
  F_{acetate}=\frac{rF_B}{1+r}
\]
\vspace{-0.16cm}
\end{block}
\end{column}
