\frametitle{\large \textbf{Henderson-Hasselbalch Model}}
\scriptsize

\vspace{-0.10cm}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.50\textwidth}
\begin{block}{Acetate-buffer relation}
\vspace{-0.12cm}
\[
  \pH = pK_a+\log_{10}\frac{[A^-]}{[HA]}
\]
\vspace{-0.20cm}
\[
  C_{HA}^{mix}=\frac{C_0\FH}{\FH+\FA+\FW},
  \qquad
  C_{A^-}^{mix}=\frac{C_0\FA}{\FH+\FA+\FW}
\]
\vspace{-0.14cm}
\end{block}

\vspace{0.04cm}
\begin{block}{Why water cancels}
\vspace{-0.12cm}
\[
  \frac{C_{A^-}^{mix}}{C_{HA}^{mix}}
  =
  \frac{C_0\FA/(\FH+\FA+\FW)}
       {C_0\FH/(\FH+\FA+\FW)}
  =
  \frac{\FA}{\FH}
\]
\vspace{-0.10cm}
\begin{itemize}\raggedright
  \setlength\itemsep{0.35ex}
  \item The same total-flow denominator appears in both buffer components.
  \item With equal 100 mM stocks, the steady-state pH ratio is acid/acetate only.
\end{itemize}
\end{block}
\end{column}

\begin{column}{0.47\textwidth}
\begin{block}{Steady-state calculation}
\vspace{-0.10cm}
\[
  \pH_{HH,k}
  =
  pK_a+\log_{10}\left(
  \frac{F_{acetate,k}}{F_{acid,k}}
  \right)
\]
\vspace{-0.10cm}
\begin{itemize}\raggedright
  \setlength\itemsep{0.35ex}
  \item Water is not a primary steady-state pH actuator.
  \item It dilutes acid and acetate together.
  \item The acid/acetate ratio is unchanged.
\end{itemize}
\end{block}

\vspace{0.04cm}
\begin{block}{Where water still matters}
\vspace{-0.12cm}
\[
  F_T=\FH+\FA+\FW,
  \qquad
  \tau \approx \frac{V}{F_T}
\]
\vspace{-0.16cm}
\[
  \Cbuf =
  \frac{C_0(\FH+\FA)}{\FH+\FA+\FW}
\]
\vspace{-0.12cm}
\begin{itemize}\raggedright
  \setlength\itemsep{0.35ex}
  \item dilution and buffer strength,
  \item hydraulic residence time and transport delay,
  \item mixing and sensor response.
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}

\frametitle{\large \textbf{Raw three-flow control is non-unique for pH tracking}}
\scriptsize

\vspace{-0.12cm}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.40\textwidth}
\begin{block}{Problem with raw actions}
\vspace{-0.12cm}
\[
  u=\begin{bmatrix}F_{acid}\\F_{acetate}\\F_W\end{bmatrix},
  \qquad
  \pH\approx pK_a+\log_{10}\frac{F_{acetate}}{F_{acid}}
\]
\vspace{-0.20cm}
\[
  (1,1),\ (3,3),\ (8,8)
  \quad\Rightarrow\quad
  F_{acetate}/F_{acid}=1

\begin{itemize}\raggedright

  \item drifting to very low flows,
  \item unstable/chattering pump actions,

\frametitle{\large \textbf{Controller structure 1: ratio-based controller}}
\scriptsize

\vspace{-0.12cm}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.40\textwidth}
\begin{block}{Reparameterize the action}
\vspace{-0.12cm}
\[
  r=\frac{F_{acetate}}{F_{acid}},
  \qquad
  z=\log_{10}\frac{F_{acetate}}{F_{acid}}
\]
\vspace{-0.18cm}
\[
  \pH \approx pK_a+z,
  \qquad
  F_B=F_{acid}+F_{acetate}
\]
\vspace{-0.16cm}
\end{block}

\vspace{0.04cm}
\begin{block}{Deterministic flow allocator}
\vspace{-0.12cm}
\[
  r=10^z,\quad
  F_{acid}=\frac{F_B}{1+r},\quad
  F_{acetate}=\frac{rF_B}{1+r}
\]
\vspace{-0.16cm}
\end{block}
\end{column}

\begin{frame}
\setbeamertemplate{blocks}[rounded][shadow=false]
\setbeamercolor{block body}{fg=black,bg=blockTint}
\frametitle{\large \textbf{Controller structure 2: chemical usage objective}}
\scriptsize

\vspace{-0.12cm}
\begin{block}{Chemical usage objective}
\vspace{-0.12cm}
\[
  \begin{aligned}
  R_k =&
  -Q(\pH_k-\pH_{sp})^2
  -\lambda_{\Delta u}\|\Delta u_k\|^2\\
  &-\lambda_B(F_B-F_{B,ref})^2
  -\lambda_{chem}(F_{acid}+F_{acetate})
  \end{aligned}
\]
\vspace{-0.18cm}
\[
  C_{buffer}\ge C_{min}
  \qquad
  C_{buffer}=
  \frac{C_0(F_{acid}+F_{acetate})}{F_{acid}+F_{acetate}+\FW}
\]
\vspace{-0.16cm}
\end{block}

\vspace{0.04cm}
\begin{block}{Clean hierarchy from the document}
\renewcommand{\arraystretch}{1.18}
\centering
\begin{tabular}{@{}l@{\quad$\rightarrow$\quad}l@{}}
pH tracking & acid/acetate ratio\\
buffer strength & total acid+acetate flow\\
dilution/residence time & water or total flow\\
\end{tabular}
\end{block}

\vspace{0.04cm}
\begin{block}{Pre-training implication}
\justifying
With this model and objective, we can start pre-training an RL agent, but we still have to decide the agent structure and the calibrated dynamic model used for training.
\end{block}
